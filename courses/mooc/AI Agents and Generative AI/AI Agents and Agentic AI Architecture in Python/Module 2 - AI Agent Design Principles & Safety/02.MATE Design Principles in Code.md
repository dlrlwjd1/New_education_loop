# MATE 설계 원칙

체스에서 체크메이트는 전략의 완벽한 실행을 의미한다. 모든 말이 올바른 위치에 배치되고, 모든 수가 최대 효과를 위해 계산되어 필연적인 승리로 이어진다. AI 에이전트를 구축할 때도 이와 유사한 전략적 사고를 MATE 원칙을 통해 적용할 수 있다: **Model efficiency(모델 효율성), Action specificity(액션 특정성), Token efficiency(토큰 효율성), Environmental safety(환경 안전성)**.

---

## Model Efficiency: 말을 현명하게 선택하라

체스에서 각 말은 고유한 능력을 가진다. 폰으로 충분한 상황에 퀸을 쓰지는 않는다. 마찬가지로, 모델 효율성이란 각 작업에 맞는 적절한 LLM을 선택하는 것을 의미한다.

```python
@register_tool(description="텍스트에서 기본 연락처 정보를 추출한다")
def extract_contact_info(action_context: ActionContext, text: str) -> dict:
    """더 작고 빠른 모델을 사용하여 텍스트에서 이름, 이메일, 전화번호를 추출한다."""
    # 단순 추출 작업에는 더 작은 모델을 사용
    response = action_context.get("fast_llm")(Prompt(messages=[
        {"role": "system", "content": "연락처 정보를 JSON 형식으로 추출하세요."},
        {"role": "user", "content": text}
    ]))
    return json.loads(response)

@register_tool(description="복잡한 기술 문서를 분석한다")
def analyze_technical_doc(action_context: ActionContext, document: str) -> dict:
    """기술 문서에 대한 심층 분석을 수행한다."""
    # 복잡한 분석에는 더 강력한 모델을 사용
    response = action_context.get("powerful_llm")(Prompt(messages=[
        {"role": "system", "content": "이 기술 문서를 철저히 분석하여 예상치 못한 문제로 이어질 수 있는 프로세스상의 잠재적 모순을 파악하세요."},
        {"role": "user", "content": document}
    ]))
    return json.loads(response)
```

## Action Specificity: 보드를 통제하라

체스에서 정밀한 위치 선정은 상대방의 선택지를 제한한다. 마찬가지로, 구체적인 액션은 오용이나 혼란의 가능성을 줄인다.

```python
# 너무 범용적 - 오용 가능성을 열어둠
@register_tool(description="캘린더 이벤트 수정")
def update_calendar(action_context: ActionContext, 
                   event_id: str,
                   updates: dict) -> dict:
    """캘린더 이벤트의 모든 속성을 업데이트한다."""
    return calendar.update_event(event_id, updates)

# 더 구체적 - 명확한 목적과 제한된 범위
@register_tool(description="내가 소유한 회의를 새 시간으로 재예약한다")
def reschedule_my_meeting(action_context: ActionContext,
                         event_id: str,
                         new_start_time: str,
                         new_duration_minutes: int) -> dict:
    """
    내가 소유한 회의를 새 시간으로 재예약한다.
    본인이 주최자인 회의에만 작동한다.
    """
    # 소유권 확인
    event = calendar.get_event(event_id)
    if event.organizer != action_context.get("user_email"):
        raise ValueError("본인이 주최한 회의만 재예약할 수 있습니다")
        
    # 새 시간이 미래인지 검증
    new_start = datetime.fromisoformat(new_start_time)
    if new_start < datetime.now():
        raise ValueError("과거 시간으로는 회의를 예약할 수 없습니다")
        
    return calendar.update_event_time(
        event_id,
        new_start_time=new_start_time,
        duration_minutes=new_duration_minutes
    )
```

## Token Efficiency: 모든 수를 최대화하라

체스에서 모든 수는 자신의 포지션을 발전시켜야 한다. 예를 들어 매출 데이터를 분석할 때 전년 대비 성장률과 상위 3개 트렌드만 필요하다면, 불필요한 컨텍스트나 과도한 지시로 토큰을 낭비해선 안 된다. 때로는 충분한 추론을 위해 토큰을 더 써야 할 수도 있지만, 항상 가장 효율적인 사용법을 테스트하고 최적화해야 한다. 프롬프트가 길수록 입력 토큰이 더 소비되고, LLM에 필요 이상의 작업을 할당하면 출력 토큰도 낭비된다.

```python
# 토큰 비효율적 - 불필요한 컨텍스트 포함
@register_tool(description="트렌드와 패턴을 파악하기 위해 매출 데이터를 분석한다...")
def analyze_sales(action_context: ActionContext, data: str) -> str:
    """
    이 함수는 매출 데이터를 분석하여 트렌드와 패턴을 파악한다.
    다음과 같은 다양한 측면을 살펴본다:
    - 월별 트렌드
    - 계절적 패턴
    - 전년 대비 성장률
    - 제품 카테고리 성과
    - 지역별 변화
    - 고객 세그먼트
    
    분석은 다양한 요소를 고려하여 철저하게 진행된다...
    [더 장황한 설명]
    """
    
    # 불필요한 분석을 요청하는 프롬프트 - 입력과 출력 토큰 모두 낭비
    return prompt_llm(action_context, f"""
        이 매출 데이터를 철저히 분석하세요. 월별 트렌드,
        계절적 패턴, 전년 대비 성장률, 제품 카테고리,
        지역별 변화, 고객 세그먼트를 고려하세요. 이 모든 측면에 대해
        상세한 인사이트를 제공하세요.
        
        데이터: {data}
        
        포괄적인 분석을 제공해 주세요...
    """)

# 토큰 효율적 - 집중적이고 정확
@register_tool(description="핵심 트렌드 파악을 위해 매출 데이터를 분석한다")
def analyze_sales(action_context: ActionContext, data: str) -> str:
    """핵심 매출 지표를 계산하고 중요한 트렌드를 파악한다."""
    
    # 집중적이고 정확한 프롬프트 - 토큰을 효율적으로 사용
    return prompt_llm(action_context, f"""
        매출 데이터: {data}
        1. 전년 대비 성장률 계산
        2. 상위 3개 트렌드 파악
        3. 주요 이상 징후 표시
    """)
```
